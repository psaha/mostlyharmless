\section{Spin and tide}

\begin{equation}
  \Phi(\mathbf r) = -\frac{1}{r}
    - \etide\, r^2 P_2(\hat{\mathbf r}\cdot \uvectide)
    + \espin\, r^2 P_2(\hat{\mathbf r}\cdot \uvecspin)
    - \espin\, r^2
  \label{eq:deformation-potential}
\end{equation}
where
\begin{equation}
  \espin = (1 + k_2)\,\frac{\Omega^{2}a^{3}}{3GM} , \qquad
  \etide = (1 + k_2)\,\frac{M'}{M}\Bigl(\frac{a}{r'}\Bigr)^{3}
  \label{eq:deformation-amplitudes}
\end{equation}

\begin{equation}
  \frac{1}{r} + E(\uvecr)\, r^2 = \frac{1}{1-\espin}
\end{equation}

\begin{equation}
  E(\uvecr) = \etide P_2(\uvecr\cdot\uvectide)
            - \espin\bigl[P_2(\uvecr\cdot\uvecspin) - 1\bigr]
  \label{eq:level-depth}
\end{equation}

\begin{equation}
  g = 1 - 2\espin
    - 4\Bigl[\etide\,P_2(\uvecr\cdot\uvectide)
    - \espin\,P_2(\uvecr\cdot\uvecspin)\Bigr]
    + O(\epsilon^{2})
  \label{eq:deformation-gravity-first-order}
\end{equation}
