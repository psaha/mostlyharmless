\section{Fitted coefficients}

\begin{equation}
\begin{aligned}
  G_{lm,l'm'} &= \iint \Yup_{lm}(\Lambda^{-1}\circ(x,y)) \;
                 \Yup_{l'm'}^*(\Lambda^{-1}\circ(x,y)) \; dx\,dy  \\
  B_{lm}      &= \iint I(x,y) \;
                 \Yup_{l'm'}^*(\Lambda^{-1}\circ(x,y)) \; dx\,dy
\end{aligned}
\end{equation}

\begin{equation}
  \sum_{l'm'} G_{lm,l'm'} \; a_{l'm'} = B_{lm}
\end{equation}
