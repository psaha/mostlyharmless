\section{Spherical harmonics}

\begin{equation}
\begin{aligned}
  Y_{lm}(\theta,\varphi) &= N_{lm}\, P_{lm}(\cos\theta)\, e^{im\varphi}  \\
  N_{lm}^2 = (2l+1)\,\frac{(l-m)!}{(l+m)!}
\end{aligned}
\label{eq:ylm}
\end{equation}
where
\begin{equation}
  P_{lm}(x) = \frac{1}{2^{l}l!} \, (1-x^{2})^{m/2} \,
              \frac{d^{l+m}}{dx^{l+m}} (x^{2}-1)^{l}
\label{eq:associated-legendre}
\end{equation}

\begin{equation}
  Y_{l,-m} =   Y_{l,m}^*
\end{equation}

\begin{equation}
\begin{matrix}
    Y_{00} = 1 \hfill
  & Y_{10} = \sqrt{3}\,\cos\theta \hfill
  & Y_{11} = \sqrt{\tfrac32}\,\sin\theta\,e^{i\varphi} \hfill \\[4pt]
    Y_{20} = \tfrac{\sqrt{5}}{2}(3\cos^{2}\theta-1) \qquad \hfill
  & Y_{21} = \sqrt{\tfrac{15}{2}}\,\sin\theta\cos\theta\,e^{i\varphi}
    \qquad \hfill
  & Y_{22} = \sqrt{\tfrac{15}{8}}\,\sin^{2}\theta\,e^{2i\varphi} \hfill
\end{matrix}
\label{eq:ylm-table}
\end{equation}

Addition theorem
\begin{equation}
  P_2(\uvecr\cdot\uvece) = \frac{1}{5}\sum_{m=-2}^{2}
                           Y_{2m}^*(\uvece)\,Y_{2m}(\uvecr)
  \label{eq:zonal-l2}
\end{equation}
