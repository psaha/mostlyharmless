\section{Spherical harmonics}

\begin{equation}
  Y_{lm}(\theta,\varphi) = N_{lm}\, P_{lm}(\cos\theta)\, e^{im\varphi}
\end{equation}
where
\begin{equation}
\begin{aligned}
  P_{lm}(x) &= (1-x^{2})^{m/2}\,\frac{d^{m}}{dx^{m}}P_l(x) \\
  P_l(x)    &= \frac{1}{2^{l}l!}\,\frac{d^{l}}{dx^{l}}\,(x^{2}-1)^{l}
\end{aligned}
\label{eq:associated-legendre}
\end{equation}
and
\begin{equation}
  N_{lm} = \sqrt{(2l+1)\,\frac{(l-m)!}{(l+m)!}} ,
  \label{eq:ylm}
\end{equation}

\begin{equation}
  Y_{l,-m} =   Y_{l,m}^*
\end{equation}

\begin{equation}
\begin{matrix}
    Y_{00} = 1 \hfill
  & Y_{10} = \sqrt{3}\,\cos\theta \hfill
  & Y_{11} = \sqrt{\tfrac32}\,\sin\theta\,e^{i\varphi} \hfill \\[4pt]
    Y_{20} = \tfrac{\sqrt{5}}{2}(3\cos^{2}\theta-1) \qquad \hfill
  & Y_{21} = \sqrt{\tfrac{15}{2}}\,\sin\theta\cos\theta\,e^{i\varphi}
    \qquad \hfill
  & Y_{22} = \sqrt{\tfrac{15}{8}}\,\sin^{2}\theta\,e^{2i\varphi} \hfill
\end{matrix}
\label{eq:ylm-table}
\end{equation}
